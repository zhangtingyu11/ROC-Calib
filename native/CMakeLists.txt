cmake_minimum_required(VERSION 3.16)
project(autocalib_native LANGUAGES C CXX)
find_package(OpenCV QUIET COMPONENTS core imgproc imgcodecs calib3d)
if(NOT OpenCV_FOUND)
  find_path(OPENCV_INCLUDE_DIR opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
  include_directories(${OPENCV_INCLUDE_DIR})
  foreach(component core imgproc imgcodecs calib3d)
    find_library(OPENCV_${component} opencv_${component} REQUIRED)
    list(APPEND OpenCV_LIBS ${OPENCV_${component}})
  endforeach()
endif()
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(BLAS REQUIRED)
find_package(LAPACK REQUIRED)
add_library(native_optimizers STATIC vendor/__lbfgsb.c vendor/__slsqp.c vendor/__nnls.c)
target_compile_options(native_optimizers PRIVATE -O3 -ffp-contract=off)
target_link_libraries(native_optimizers PUBLIC ${BLAS_LIBRARIES} ${LAPACK_LIBRARIES})
add_executable(autocalib_native main.cpp data.cpp geometry.cpp objective.cpp solver.cpp cauchy.cpp vendor/sqpnp.cpp)
target_compile_features(autocalib_native PRIVATE cxx_std_17)
target_compile_options(autocalib_native PRIVATE -O3 -ffp-contract=off)
target_link_libraries(autocalib_native PRIVATE Eigen3::Eigen ${OpenCV_LIBS} OpenMP::OpenMP_CXX OpenSSL::Crypto native_optimizers)
add_executable(check_skip check.cpp data.cpp geometry.cpp objective.cpp reference_objective.cpp cauchy.cpp vendor/sqpnp.cpp)
target_compile_features(check_skip PRIVATE cxx_std_17)
target_compile_options(check_skip PRIVATE -O3 -ffp-contract=off)
target_link_libraries(check_skip PRIVATE Eigen3::Eigen ${OpenCV_LIBS} OpenMP::OpenMP_CXX OpenSSL::Crypto)
add_executable(check_gradient check_gradient.cpp data.cpp geometry.cpp objective.cpp cauchy.cpp vendor/sqpnp.cpp)
target_compile_features(check_gradient PRIVATE cxx_std_17)
target_compile_options(check_gradient PRIVATE -O3 -ffp-contract=off)
target_link_libraries(check_gradient PRIVATE Eigen3::Eigen ${OpenCV_LIBS} OpenMP::OpenMP_CXX OpenSSL::Crypto)
add_executable(check_support check_support.cpp)
target_compile_features(check_support PRIVATE cxx_std_17)
target_link_libraries(check_support PRIVATE Eigen3::Eigen ${OpenCV_LIBS})
